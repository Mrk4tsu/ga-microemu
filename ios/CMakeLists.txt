cmake_minimum_required(VERSION 3.20)
project(MicroEmuHost C)
set(CMAKE_C_STANDARD 11)
set(MINIJVM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/miniJVM" CACHE PATH "Pinned miniJVM source")
set(CORE "${MINIJVM_ROOT}/minijvm/c")
file(GLOB JVM_SOURCES CONFIGURE_DEPENDS "${CORE}/jvm/*.c" "${CORE}/utils/*.c"
     "${CORE}/utils/https/*.c" "${CORE}/utils/https/mbedtls/library/*.c")
add_library(microemu_core STATIC ${JVM_SOURCES} "${CORE}/utils/mimalloc/src/static.c"
            native/bridge.c native/raster.c)
target_include_directories(microemu_core PUBLIC native "${CORE}/jvm" "${CORE}/utils" "${CORE}/utils/sljit"
    "${CORE}/utils/https" "${CORE}/utils/https/mbedtls/include" "${CORE}/utils/mimalloc/include"
    "${MINIJVM_ROOT}/mobile/c/gui/stb")
if(MSVC)
    target_compile_options(microemu_core PRIVATE /utf-8 /wd4996)
    target_compile_definitions(microemu_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_link_libraries(microemu_core PUBLIC pthread m)
    if(NOT APPLE)
        target_link_libraries(microemu_core PUBLIC dl)
    endif()
endif()
add_executable(microemu-headless tests/headless.c)
target_link_libraries(microemu-headless PRIVATE microemu_core)
if(WIN32)
    target_link_libraries(microemu-headless PRIVATE ws2_32)
endif()
enable_testing()
add_executable(raster-test tests/raster_test.c native/raster.c)
target_include_directories(raster-test PRIVATE native)
if(NOT MSVC)
    target_link_libraries(raster-test PRIVATE m)
endif()
add_test(NAME raster COMMAND raster-test)
